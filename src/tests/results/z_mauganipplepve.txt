        玩家造成伤害;
        双方;
        末日铁拳;
    }
    条件 {
        "or (obj == OBJ.MORE and victim in objPos)"
        被攻击方 == 全局.nipman;
        全局.obj != -1;
        事件技能 == 按钮(辅助攻击模式);
    }
    动作 {
        设置状态(事件玩家, 空, 被入侵, 0.5);
        清除状态(被攻击方, 冰冻);
        等待(0.48, 当为“假”时中止);
        设置状态(被攻击方, 空, 冰冻, 9999);
    }
}

}

}

规则 ("sombra aint slick") {
    事件 {
        持续 - 每名玩家;
        双方;
        黑影;
    }
    条件 {
        或(正在使用终极技能(事件玩家), 与(按钮被按下(事件玩家, 按钮(终极技能)), 比较(终极技能充能百分比(事件玩家), ==, 100))) == 真;
    }
    动作 {
        开始伤害调整(选择英雄的玩家(英雄(毛加), 队伍2), 事件玩家, 2, 无);
        设置玩家变量(事件玩家, mobTarget, 上一个伤害调整ID);
        等待直到 (比较(终极技能充能百分比(事件玩家), <, 100), 2);
        停止伤害调整((事件玩家).mobTarget);
    }
}
