规则 ("[Setup] init constants & starting variables") {
    事件 {
        持续 - 全局;
    }
    动作 {
        "Check if icons are used as symbols"
        If(全局.SETTING_ICON_CARDS);
            "Set the number of cards to 10"
            设置全局变量(SETTING_AMOUNT_OF_NUMBERS, 10);
        End;
        "Give the first turn at least one minute to play"
        设置全局变量(TurnTimeRemaining, If-Then-Else(全局.SETTING_TURN_TIME_LIMIT, 较大(60, 全局.SETTING_TURN_TIME_LIMIT), 0));
        "Start chasing the turn time remaining variable to 0 when the match starts"
        追踪全局变量频率(TurnTimeRemaining, 0, 游戏正在进行中, 速率及最终值);
        "Setup constants"
        设置全局变量(COLORS, 已过滤的数组(数组(字符串("红色"), 自定义字符串("Orange"), 字符串("黄色"), 自定义字符串("Lime Green"), 字符串("绿色"), 自定义字符串("Turquoise"), 自定义字符串("Aqua"), 自定义字符串("Sky Blue"), 字符串("蓝色"), 字符串("亮紫色")), 数组包含(数组分割(全局.SETTING_COLOR_ORDER, 0, 全局.SETTING_AMOUNT_OF_COLORS), 当前数组元素)));
        设置全局变量(WILD_CARD_SYMBOLS, 数组(自定义字符串("+4"), 自定义字符串("Wild")));
        设置全局变量(PENALTY_CARD_SYMBOLS, 数组(自定义字符串("+2"), 自定义字符串("+4")));
        设置全局变量(WHITE_CARD_SYMBOLS, 数组(自定义字符串("+4"), 自定义字符串("×2"), 自定义字符串("Copycat"), 自定义字符串("Shuffle"), 自定义字符串("Swap-All"), 自定义字符串("Wild")));
        设置全局变量(NUMBER_CARD_SYMBOLS, 空数组);
        设置全局变量(COLORED_CARD_SYMBOLS, 空数组);
        "Check if icon cards are enabled"
        If(非(全局.SETTING_ICON_CARDS));
            "Add every possible number"
            For 全局变量(TempSymbolIndex, 0, 全局.SETTING_AMOUNT_OF_NUMBERS, 1);
                修改全局变量(NUMBER_CARD_SYMBOLS, 添加至数组, 自定义字符串("{0}", 全局.TempSymbolIndex));
                修改全局变量(COLORED_CARD_SYMBOLS, 添加至数组, 自定义字符串("{0}", 全局.TempSymbolIndex));
            End;
            设置全局变量(SPECIAL_CARD_SYMBOLS, 数组(自定义字符串("+2"), 自定义字符串("+4"), 自定义字符串("×2"), 自定义字符串("Copycat"), 自定义字符串("Reverse"), 自定义字符串("Shuffle"), 自定义字符串("Skip"), 自定义字符串("Swap"), 自定义字符串("Swap-All"), 自定义字符串("Wild")));
        Else;
            修改全局变量(COLORED_CARD_SYMBOLS, 添加至数组, 数组(0, 图标字符串(剧毒), 图标字符串(满月), 图标字符串(火焰), 图标字符串(晕眩), 图标字符串(箭矢), 图标字符串(红桃), 图标字符串(方块), 图标字符串(黑桃), 图标字符串(梅花)));
        End;
        "Setup helper variables for reverse card effects"
        设置全局变量(DirectionOfPlay, -1);
        追踪全局变量频率(DirectionOfPlayTimer, 0, 1.5, 速率及最终值);
        "Set a variable for copycat messages"
        设置全局变量(LastPlayedCardPlayer, 自定义字符串("the discard pile"));
        设置全局变量(BoundaryRadius, 0.95);
        设置全局变量(BoundaryForce, 0.85);
    }
}

规则 ("[Setup] create deck") {
    事件 {
        持续 - 全局;
    }
    动作 {
        "Initialize the arrays"
        设置全局变量(DeckSymbols, 空数组);
        设置全局变量(DeckColors, 空数组);
        "Loop once for every color"
        For 全局变量(TempColorIndex, 0, 数量(全局.COLORS), 1);
            "Add all cards for the current color"
            调用子程序(AddColoredSymbolsToDeck);
            "Set the color of all newly added symbols to the current color in the loop"
            While(比较(数量(全局.DeckColors), <, 数量(全局.DeckSymbols)));
                修改全局变量(DeckColors, 添加至数组, 数组中的值(全局.COLORS, 全局.TempColorIndex));
            End;
        End;
        "Add all uncolored symbols"
        调用子程序(AddUncoloredSymbolsToDeck);
        "Set the color of all newly added symbols to be uncolored"
        While(比较(数量(全局.DeckColors), <, 数量(全局.DeckSymbols)));
            修改全局变量(DeckColors, 添加至数组, 字符串("白色"));
        End;
        "Select a random index of a card in the deck to use as the starting card"
        设置全局变量(TempNumber, 随机整数(0, 减(数量(全局.DeckSymbols), 1)));
        "Set the starting card to the selected card"
        设置全局变量(DiscardPileSymbol, 数组中的值(全局.DeckSymbols, 全局.TempNumber));
        设置全局变量(DiscardPileColor, 数组中的值(全局.DeckColors, 全局.TempNumber));
        设置全局变量(DiscardPileLastColor, 全局.DiscardPileColor);
        "Remove the starting card from the deck"
        修改全局变量(DeckSymbols, 根据索引从数组中移除, 全局.TempNumber);
        修改全局变量(DeckColors, 根据索引从数组中移除, 全局.TempNumber);
        "Save a copy of the created deck"
        设置全局变量(DeckSymbolsCopy, 全局.DeckSymbols);
        设置全局变量(DeckColorsCopy, 全局.DeckColors);
    }

规则 ("[Effects] update discard pile color (turquoise)") {
    事件 {
        持续 - 全局;
    }
    条件 {
        "The color of the discard pile card has changed to turquoise"
        全局.DiscardPileColor == 自定义字符串("Turquoise");
    }
    动作 {
        "Destroy the previous symbol effect"
        消除地图文本(全局.DiscardPileSymbolTextID);
        创建地图文本(所有玩家(所有队伍), If-Then-Else(与(全局.DiscardPileSymbolMovementTimer, 比较(最后(全局.DeckSymbols), ==, 自定义字符串("Copycat"))), 自定义字符串("Copycat"), If-Then-Else(与(数组包含(全局.PENALTY_CARD_SYMBOLS, 全局.DiscardPileSymbol), 非(全局.DiscardPileSymbolMovementTimer)), 自定义字符串("{0} (+{1})", 全局.DiscardPileSymbol, 全局.CurrentPenalty), If-Then-Else(比较(全局.DiscardPileSymbol, ==, 自定义字符串("Wild")), 自定义字符串("Turquoise"), 全局.DiscardPileSymbol))), 加(加(乘(减(1, 全局.DiscardPileSymbolMovementTimer), 全局.MapPosition), 乘(全局.DiscardPileSymbolMovementTimer, 全局.LastPlayedCardPosition)), 乘(上, 加(加(1.5, 乘(0.2, 角度的余弦值(乘(总计消耗时间, 36)))), 乘(3, 减(1, 乘方(绝对值(减(乘(2, 全局.DiscardPileSymbolMovementTimer), 1)), 2)))))), 10, 不要截取, 可见，位置和字符串, 颜色(青绿色), 始终可见);
        设置全局变量(DiscardPileSymbolTextID, 上一个文本ID);
        "Wait for the animation to finish"
        等待(1, 无视条件);
        "Destroy all existing discard pile color effects"
        调用子程序(DestroyDiscardPileCardEffects);

规则 ("[Subroutine] Sort cards") {
    事件 {
        子程序;
        SortHandCards;
    }
    动作 {
        "Reset the selected card index"
        设置玩家变量(事件玩家, SelectedCardIndex, 0);
        "Find all wild symbols"
        设置玩家变量(事件玩家, TempSymbols, 已过滤的数组((事件玩家).HandSymbols, 数组包含(添加至数组(自定义字符串("Copycat"), 全局.WILD_CARD_SYMBOLS), 当前数组元素)));
        "Sort the wild symbols, and add them to the temporary sorted symbol array"
        设置玩家变量(事件玩家, TempSortedHandSymbols, 已排序的数组((事件玩家).TempSymbols, 数组值的索引(添加至数组(自定义字符串("Copycat"), 全局.WILD_CARD_SYMBOLS), 当前数组元素)));
        "Reset the sorted hand colors temporary array"
        设置玩家变量(事件玩家, TempSortedHandColors, 空数组);
        "Loop until all newly added cards have been assigned a color"
        While(比较(数量((事件玩家).TempSortedHandSymbols), >, 数量((事件玩家).TempSortedHandColors)));
            修改玩家变量(事件玩家, TempSortedHandColors, 添加至数组, 字符串("白色"));
        End;
        "Loop over every color"
        For 全局变量(TempColorIndex, 0, 数量(全局.COLORS), 1);
            "Reset the temporary array"
            设置玩家变量(事件玩家, TempSymbols, 空数组);
            "Loop over all symbols the player has (can't use a filtered array here, since the index is needed to check the color but 'index of' always returns the first matching card)"
            For 全局变量(TempSymbolIndex, 0, 数量((事件玩家).HandSymbols), 1);
                "Check if the color of the current card being looped over is the current color being looped over, and the card is not a wild card"
                If(与(比较(数组中的值((事件玩家).HandColors, 全局.TempSymbolIndex), ==, 数组中的值(全局.COLORS, 全局.TempColorIndex)), 非(数组包含(添加至数组(自定义字符串("Copycat"), 全局.WILD_CARD_SYMBOLS), 数组中的值((事件玩家).HandSymbols, 全局.TempSymbolIndex)))));
                    修改玩家变量(事件玩家, TempSymbols, 添加至数组, 数组中的值((事件玩家).HandSymbols, 全局.TempSymbolIndex));
                    修改玩家变量(事件玩家, TempSortedHandColors, 添加至数组, 数组中的值(全局.COLORS, 全局.TempColorIndex));
                End;
            End;
            修改玩家变量(事件玩家, TempSortedHandSymbols, 添加至数组, 已排序的数组((事件玩家).TempSymbols, 数组值的索引(全局.COLORED_CARD_SYMBOLS, 当前数组元素)));
            "Loop until all newly added cards have been assigned a color"
            While(比较(数量((事件玩家).TempSortedHandSymbols), >, 数量((事件玩家).TempSortedHandColors)));
                修改玩家变量(事件玩家, TempSortedHandColors, 添加至数组, 数组中的值(全局.COLORS, 全局.TempColorIndex));
            End;
        End;
        "Set the player's cards to the sorted arrays"
        设置玩家变量(事件玩家, HandSymbols, (事件玩家).TempSortedHandSymbols);
        设置玩家变量(事件玩家, HandColors, (事件玩家).TempSortedHandColors);
        "Clear the hand HUD text elements of the event player"
        设置全局变量(PlayerToClearHand, 事件玩家);
        调用子程序(ClearHandHUDTexts);
        "Add the player's cards to the card render queue"
        设置玩家变量(事件玩家, HandRenderQueueSymbols, (事件玩家).HandSymbols);
        设置玩家变量(事件玩家, HandRenderQueueColors, (事件玩家).TempSortedHandColors);
    }
}
