变量 {
    全局:
        0: A
        1: B
        2: C
        3: D
        8: I
    玩家:
        8: I
}
//Optimize for size enabled
规则 ("while loop") {
    事件 {
        持续 - 全局;
    }
    动作 {
        While(比较(全局.A, ==, 全局.B));
            While(比较(全局.B, ==, 全局.C));
                修改全局变量(A, 加, 真);
                If(全局.D);
                    中断;
                End;
            End;
        End;
        修改全局变量(B, 加, 真);
    }
}

规则 ("while to the end") {
    事件 {
        持续 - 全局;
    }
    动作 {
        While(比较(全局.A, ==, 全局.B));
            根据条件跳过(全局.C, 真);
            修改全局变量(A, 加, 真);
            //__label_continue_2__:
        End;
    }
}

规则 ("while to the end nested") {
    事件 {
        持续 - 全局;
    }
    动作 {
        While(比较(全局.A, ==, 全局.B));
            While(比较(全局.B, ==, 全局.C));
                修改全局变量(A, 加, 真);
            End;
        End;
    }
}

规则 ("while false") {
    事件 {
        持续 - 全局;
    }
    动作 {
        修改全局变量(B, 加, 真);
    }
}

规则 ("for loop") {
    事件 {
        持续 - 全局;
    }
    动作 {
        For 全局变量(I, 1, 3, 2);
            修改全局变量(A, 加, 真);
        End;
        For 全局变量(I, 1, 3, 真);
            修改全局变量(A, 加, 真);
        End;
        For 全局变量(I, 0, 3, 真);
            修改全局变量(A, 加, 真);
        End;
        For 玩家变量(主机玩家, I, 0, 3, 真);
            修改全局变量(A, 加, 真);
        End;
    }
}

规则 ("do/while") {
    事件 {
        持续 - 全局;
    }
    动作 {
        修改全局变量(A, 加, 真);
        跳过(真);
        根据条件循环(比较(全局.A, ==, 全局.B));
        //__label_break_3__:
        修改全局变量(B, 加, 真);
    }
}

规则 ("nested do/while") {
    事件 {
        持续 - 全局;
    }
    动作 {
        修改全局变量(A, 加, 真);
        跳过(真);
        循环;
        //__label_break_4__:
        跳过(真);
        根据条件循环(比较(全局.A, ==, 全局.B));
        //__label_break_5__:
        修改全局变量(B, 加, 真);
    }
}

规则 ("do/while rule condition") {
    事件 {
        持续 - 全局;
    }
    条件 {
        全局.A == 全局.B;
    }
    动作 {
        等待(假, 无视条件);
        修改全局变量(A, 加, 真);
        If(比较(全局.A, >, 玩家数量(队伍2)));
            设置全局变量(A, 空);
        End;
        如条件为“真”则循环;
    }
}
