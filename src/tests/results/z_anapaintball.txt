        全部;
    }
    条件 {
        是否是机器人(事件玩家) == 假;
    }
    动作 {
        开始为机器人强制设置名称(事件玩家, 自定义字符串("​ [ -C°▥°]-C Ana Bot #{0}", 数量(已过滤的数组(所有玩家(所有队伍), (当前数组元素).IsAIBot))));
        If(比较(自定义字符串("{0}", 事件玩家), ==, 自定义字符串("​ [ -C°▥°]-C Ana Bot #{0}", 数量(已过滤的数组(所有玩家(所有队伍), (当前数组元素).IsAIBot)))));

            设置玩家变量(事件玩家, IsAIBot, 真);
        Else;
            设置玩家变量(事件玩家, IsPlayer, 真);
            设置玩家变量(事件玩家, IsAIBot, 假);
            停止为机器人强制设置名称(事件玩家);
        End;
        设置最大复生时间(事件玩家, 全局.CUSTOM_RESPAWN_TIME);
        If(全局.MatchEnded);
            禁用按钮(事件玩家, 按钮(主要攻击模式));
